## Modular - REG-Linux
#
# A modular file that is sourced for specific script lines required by ports running on REG-Linux.
#
# usage `[ -f "${controlfolder}/mod_${CFW_NAME}.txt" ] && source "${controlfolder}/mod_${CFW_NAME}.txt"`
#
# No PipeWire setup for 32-bit ports: REG has no /usr/lib32. On its musl boards 32-bit
# ports run on the armhf glibc sidecar, whose audio goes through ALSA's pulse plugin.

# Use for Godot 2
GODOT2_OPTS="-r ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"

# Use for Godot 3+
GODOT_OPTS="--resolution ${DISPLAY_WIDTH}x${DISPLAY_HEIGHT} -f"

# Godot 3's GLES3 renderer shows nothing with Zink on the PowerVR Vulkan driver
# (A133 GE8300): use its GLES2 renderer there.
if [ "$GPU_DRIVER" = "powervr" ]; then
    GODOT_OPTS="$GODOT_OPTS --video-driver GLES2"
fi

# Weston ports (westonwrap sources this file too) on bare KMS/DRM with Mesa:
# westonwrap always runs Weston's DRM backend through crusty, a fake DRM/GBM
# that shows Weston in an SDL2 window, made for Mali blobs without GBM. With
# Mesa it fails ("failed to get driver name", EGL not initialized). Run
# Weston and seatd without it, on Mesa's own GBM and EGL. Weston then needs
# the libwayland-client that Mesa's libEGL was built with (its own is older)
# and libGLESv2 (its gl-renderer relied on crusty's libEGL for GL symbols).
# westonwrap never sets $strace and $straceseat: they prefix both commands.
# Weston itself is always aarch64 (westonwrap uses lib_aarch64); on musl
# boards its libraries come from the aarch64 glibc sidecar.
if [ "${0##*/}" = "westonwrap.sh" ] && [ "$XDG_SESSION_TYPE" = "drm" ] &&
   [ "$GPU_DRIVER" != "mali" ]; then
    reg_gllib=/usr/lib
    [ -d /usr/glibc/lib ] && reg_gllib=/usr/glibc/lib
    strace="env -u LD_PRELOAD"
    strace="$strace LD_PRELOAD=$reg_gllib/libwayland-client.so.0:$reg_gllib/libGLESv2.so.2"
    strace="$strace LD_LIBRARY_PATH=$(dirname "$0")/lib_aarch64"
    straceseat="env -u LD_PRELOAD"
fi

# Weston ports under sway (XDG_SESSION_TYPE=wayland). Weston's DRM backend
# cannot take the display from sway (no DRM master, no VT), so:
#  - drm backend: Weston runs as a fullscreen sway client (its Wayland
#    backend). The port keeps Weston's kiosk shell and its XWayland, so X11
#    ports also work on images without XWayland.
#  - headless backend (crusty): Weston only hosts XWayland; crusty's SDL
#    window goes to sway instead of the invisible headless Weston.
# westonwrap moves XDG_RUNTIME_DIR to its own dir, so sway's socket and the
# Pulse socket are given by path. Weston, its XWayland and crusty need Mesa's
# libwayland-client (the pack's is older than the one Mesa's libEGL was built
# with) and libGLESv2. The pack's wayland-backend.so links libmali.so.1 (it
# was built for ROCKNIX's Mali blob, which REG never ships): that name points
# at Mesa's libEGL. westonwrap's libEGL probe for crusty fails on the same
# libwayland clash, so crusty is given Mesa's libEGL directly.
if [ "${0##*/}" = "westonwrap.sh" ] && [ "$XDG_SESSION_TYPE" = "wayland" ] &&
   { [ "$1" = "drm" ] || [ "$1" = "headless" ]; }; then
    reg_gllib=/usr/lib
    [ -d /usr/glibc/lib ] && reg_gllib=/usr/glibc/lib
    case "$WAYLAND_DISPLAY" in
        /*) reg_host_wl=$WAYLAND_DISPLAY ;;
        *)  reg_host_wl=$XDG_RUNTIME_DIR/${WAYLAND_DISPLAY:-wayland-1} ;;
    esac
    export WAYLAND_DISPLAY=$reg_host_wl
    export PULSE_RUNTIME_PATH=${PULSE_RUNTIME_PATH:-$XDG_RUNTIME_DIR/pulse}
    if [ "$1" = "drm" ]; then
        set -- wayland "${@:2}"
    else
        eval "reg_westonwrap_$(declare -f run_user_command)"
        run_user_command() {
            waydisp=$reg_host_wl
            export LD_PRELOAD=$reg_gllib/libwayland-client.so.0${LD_PRELOAD:+:$LD_PRELOAD}
            reg_westonwrap_run_user_command
        }
    fi
    [ -n "$CRUSTY_LIBEGL" ] || export CRUSTY_LIBEGL=$reg_gllib/libEGL.so.1
    # crusty asks for GLES 3.2: Mesa gives Panfrost and V3D 3.1, Lima and VC4 2.0
    if [ -z "$CRUSTY_GLES" ]; then
        case "$GPU_DRIVER" in
            panfrost|v3d) export CRUSTY_GLES=31 ;;
            lima|vc4)     export CRUSTY_GLES=20 ;;
        esac
    fi
    mkdir -p /tmp/reg-weston-egl && ln -sf "$reg_gllib/libEGL.so.1" /tmp/reg-weston-egl/libmali.so.1
    strace="env -u LD_PRELOAD"
    strace="$strace LD_PRELOAD=$reg_gllib/libwayland-client.so.0:$reg_gllib/libwayland-egl.so.1:$reg_gllib/libEGL.so.1:$reg_gllib/libGLESv2.so.2"
    strace="$strace LD_LIBRARY_PATH=$(dirname "$0")/lib_aarch64:/tmp/reg-weston-egl"
    straceseat="env -u LD_PRELOAD"
fi

pm_platform_helper() {
    pm_message_close

    if [ -e "$PM_PIPE" ]; then
        PortMasterDialogExit
    fi

    # REG exports SDL_VIDEO_DRIVER (SDL3's name) on bare KMS/DRM, and SDL3 only
    # falls back to SDL2's SDL_VIDEODRIVER when it is unset: a port's own
    # SDL_VIDEODRIVER (Dusklight's SDL3 shim: "sdl2") would be ignored.
    if [ -n "$SDL_VIDEODRIVER" ]; then
        export SDL_VIDEO_DRIVER="$SDL_VIDEODRIVER"
    fi

    # On bare KMS/DRM only one program can be DRM master: a port's loading
    # splash (tools/splash, started in the background just before) makes the
    # game's SDL2 skip the device ("kmsdrm not available"). Stop it first.
    if [ "$XDG_SESSION_TYPE" = "drm" ] && pkill -x splash 2>/dev/null; then
        i=0
        while pgrep -x splash >/dev/null 2>&1 && [ "$i" -lt 20 ]; do
            sleep 0.1
            i=$((i + 1))
        done
    fi
}
